// Power BI DAX Measures Blueprint (Import directly into Power BI Desktop)

Total Customers = COUNTROWS('telco_customers')

Retained Customers = CALCULATE(COUNTROWS('telco_customers'), 'telco_customers'[Churn] = "No")

Churned Customers = CALCULATE(COUNTROWS('telco_customers'), 'telco_customers'[Churn] = "Yes")

Churn Rate Pct = DIVIDE([Churned Customers], [Total Customers], 0)

Monthly Revenue at Risk = CALCULATE(SUM('telco_customers'[MonthlyCharges]), 'telco_customers'[Churn] = "Yes")

Annualized Revenue Lost = [Monthly Revenue at Risk] * 12
